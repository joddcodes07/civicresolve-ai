SELECT AVG(EXTRACT(EPOCH FROM ("resolvedAt" - "createdAt")) / 3600) AS avg_hours,
       MIN("resolvedAt" - "createdAt") AS fastest,
       MAX("resolvedAt" - "createdAt") AS slowest
FROM "Complaint" WHERE "resolvedAt" IS NOT NULL;
